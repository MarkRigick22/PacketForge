cmake_minimum_required(VERSION 3.16)
project(PacketForge VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PACKETFORGE_ASAN "AddressSanitizer" OFF)
option(PACKETFORGE_UBSAN "UndefinedBehaviorSanitizer" OFF)
option(PACKETFORGE_TSAN "ThreadSanitizer (separate build)" OFF)
option(PACKETFORGE_PCAP "Built-in classic PCAP reader (no libpcap required)" ON)
option(PACKETFORGE_TAP "Linux TAP adapter" ON)
if(PACKETFORGE_TSAN AND (PACKETFORGE_ASAN OR PACKETFORGE_UBSAN))
 message(FATAL_ERROR "Use TSAN in a separate build")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 add_compile_options(-Wall -Wextra -Wpedantic)
 foreach(s ASAN UBSAN TSAN)
  if(PACKETFORGE_${s})
   if(s STREQUAL "ASAN")
    set(flag address)
   elseif(s STREQUAL "UBSAN")
    set(flag undefined)
   else()
    set(flag thread)
   endif()
   add_compile_options(-fsanitize=${flag} -fno-omit-frame-pointer)
   add_link_options(-fsanitize=${flag})
  endif()
 endforeach()
endif()
find_package(Threads REQUIRED)
add_library(cnet cnet/ethernet.c cnet/ipv4.c cnet/tcp.c cnet/udp.c cnet/packet.c)
target_include_directories(cnet PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(packetforge_core src/memory/memory_pool.cpp src/device/device.cpp src/concurrency/worker.cpp src/dataplane.cpp src/baseline.cpp src/traffic.cpp src/pcap/pcap_reader.cpp src/tap/tap.cpp)
target_include_directories(packetforge_core PUBLIC include)
target_link_libraries(packetforge_core PUBLIC cnet Threads::Threads)
target_compile_definitions(packetforge_core PUBLIC PF_PCAP=$<BOOL:${PACKETFORGE_PCAP}>)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND PACKETFORGE_TAP)
 target_compile_definitions(packetforge_core PUBLIC PF_TAP=1)
else()
 target_compile_definitions(packetforge_core PUBLIC PF_TAP=0)
 message(STATUS "TAP unavailable: requires Linux and PACKETFORGE_TAP=ON")
endif()
add_executable(packetforge src/main.cpp)
target_link_libraries(packetforge PRIVATE packetforge_core)
foreach(b baseline optimized scaling)
 add_executable(benchmark_${b} benchmarks/benchmark_${b}.cpp)
 target_link_libraries(benchmark_${b} PRIVATE packetforge_core)
endforeach()
enable_testing()
add_executable(packetforge_tests tests/test_main.cpp tests/test_ring.cpp tests/test_pool.cpp tests/test_parsers.cpp tests/test_rss.cpp tests/test_dataplane.cpp tests/test_concurrency.cpp tests/test_faults.cpp tests/test_pcap.cpp)
target_link_libraries(packetforge_tests PRIVATE packetforge_core)
add_test(NAME packetforge_tests COMMAND packetforge_tests)
set_tests_properties(packetforge_tests PROPERTIES TIMEOUT 90)
add_test(NAME cli_smoke COMMAND packetforge --synthetic --packets 1000 --workers 2)
add_test(NAME baseline_smoke COMMAND packetforge --synthetic --baseline --packets 1000 --workers 2)
